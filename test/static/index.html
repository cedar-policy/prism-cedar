// Annotation ::= '@' ANYIDENT ( '('STR')' )?
// see https://docs.cedarpolicy.com/policies/syntax-grammar.html#grammar-annotation
@id("policy0")
@advice("contact support")
permit (principal, action, resource);

// the value is optional
@doNotOptimize
permit (principal, action, resource);

// ANYIDENT, not IDENT, so reserved words are legal annotation names
@is("reserved")
@if("reserved")
@true("reserved")
@__cedar("reserved")
forbid (principal, action, resource);

// an annotation value is a string, and a '@' inside a string is not an annotation
@doc("see @id and https://docs.cedarpolicy.com/")
permit (principal, action, resource)
when { context.note == "@notAnAnnotation" };

datetime.cedar


// datetime(), offset, durationSince, toDate, toTime
// duration(), toMilliseconds, toSeconds, toMinutes, toHours, toDays
// see https://github.com/cedar-policy/rfcs/blob/main/text/0080-datetime-extension.md
permit (principal, action, resource)
when
{
  datetime("2024-08-21T") == datetime("2024-08-21T00:00:00.000Z") && duration("1d2h3m4s5ms").toSeconds() == 0 &&
  datetime("2024-08-21T") == datetime("2024-08-21T23:59:59.999Z").toDate() && 
  datetime("2024-08-21T").offset(duration("1d")) == datetime("2024-08-22T") &&
  datetime("2024-08-21T").toTime() == datetime("2024-08-21T").durationSince(datetime("2024-08-21T").toDate())
};

permit (principal, action, resource)
when
{
  duration("1d").toDays() * 24 == duration("1d").toHours() &&
  duration("1h").toHours() * 60 == duration("1h").toMinutes() &&
  duration("1m").toMinutes() * 60 == duration("1m").toSeconds() &&
  duration("1s").toSeconds() * 1000 == duration("1s").toMilliseconds() &&
  duration("1ms").toMilliseconds() == 1
};

entities.cedar


// Entity ::= Path '::' STR
// Path   ::= IDENT {'::' IDENT}
// An entity literal always ends in a string; a bare Path is a type or an ExtFun.
permit (principal, action, resource)
when { principal == User::"alice" }
when { principal == ExampleCo::User::"alice" }
when { resource in A::B::C::Folder::"shared" }
when { action == Action::"view" };

// a Path in `is` position is a type, not an entity literal
permit (
  principal is ExampleCo::User,
  action in [Action::"view", Action::"edit"],
  resource is Photo
);

// comments are recognized even directly after a path separator
permit (principal, action, resource)
when { Namespace::User::"alice" == principal };// no space before this comment

// poorly formed: `__cedar` is reserved everywhere in a policy, and no qualified
// ExtFun call is valid, so neither `__cedar` nor `decimal` here forms a Path --
// this checks that the leading segment is not scoped as an entity type
permit (principal, action, resource)
when { __cedar::decimal("1.0").lessThan(decimal("2.0")) };

methods.cedar


// Access ::= '.' IDENT ['(' [ExprList] ')'] | '[' STR ']'
// ExtFun ::= [Path '::'] IDENT
// A method name is only a method after a receiver dot.
permit (principal, action, resource)
when { context.tags.contains("a") }
when { context.tags.containsAll(["a", "b"]) }
when { context.tags.containsAny(["a", "b"]) }
when { context.tags.isEmpty() }
when { resource.hasTag("write") }
when { resource.getTag("write") == "yes" };

// the same names used as attributes are not method calls
permit (principal, action, resource)
when { context.contains == 1 }
when { context.isEmpty == 2 }
when { context.offset == 3 }
when { context.toDays == 4 };

// a bare call is an ExtFun call, not a method call
permit (principal, action, resource)
when { ip("127.0.0.1").isInRange(ip("127.0.0.0/8")) }
when { decimal("1.0").lessThan(decimal("2.0")) }
when { datetime("2024-10-15").offset(duration("1d")) > datetime("2024-10-15") }
when { duration("1d").toHours() == 24 };

// index access uses a string key
permit (principal, action, resource)
when { context["owner info"].name == "Alice" };

numbers.cedar


// INT ::= '-'? ['0'-'9']+
// see https://docs.cedarpolicy.com/policies/syntax-grammar.html#grammar-int
permit (principal, action, resource)
when { context.zero == 0 }
when { context.n == 42 }
when { context.max == 9223372036854775807 }
// leading zeros are a single INT per the grammar
when { context.padded == 007 }

attributes.cedarschema


// AttrDecls := Annotations Name ['?'] ':' Type [',' | ',' AttrDecls]
// Name      := IDENT | STR
// see https://docs.cedarpolicy.com/schema/human-readable-schema-grammar.html
namespace NS {
  // a quoted attribute name is a Name, not a string value
  type Quoted = {
    "id": Long,
    "name": String,
    "owner info": String,
  };

  // optional attributes
  entity Resource {
    metadata?: String,
    required: Long,
    "quoted optional"?: Bool
  };

  // words that are keywords elsewhere are legal attribute names
  type Names = {
    "type": __cedar::String,
    namespace: String,
    entity: String,
    action: String,
    enum: Long,
    tags: Bool,
    Set: Long,
    appliesTo: Bool,
    // `in` is a reserved identifier, so it is only legal quoted
    "in": String
  };

  // an ordinary string containing a colon is still a string
  @doc("id: the task id")
  entity Doc = {
    url: String
  };
}

declarations.cedarschema


// Namespace := (Annotations 'namespace' Path '{' {Decl} '}') | Decl
// Entity    := ... 'entity' Idents ['in' EntOrTyps] [['='] RecType] ['tags' Type] ';'
//            | ... 'entity' Idents 'enum' '[' STR+ ']' ';'
// see https://docs.cedarpolicy.com/schema/human-readable-schema-grammar.html

// a namespace name is a Path, not a single IDENT
namespace Foo::Bar::Baz {
  entity E;
}

// declarations are not line oriented
namespace M { entity U; action x appliesTo { principal: U, resource: U }; }

// 'in' takes a single EntType or a bracketed list
entity Album, Folder, Drive;
entity Photo in Album;
entity Document in [Folder, Drive];
action "read", "write", "admin";
action view in Action::"read";
action edit in [Action::"write", Action::"admin"];

// enumerated entity types
entity Color enum ["Red", "Blue", "Green"];
entity Status enum ["active"];

// the '=' before a record shape is optional
entity WithEquals = { a: Long };
entity WithoutEquals { a: Long };

// appliesTo may be omitted entirely, and may use bracketed or bare entity types
action a1;
action a2 appliesTo { principal: [Foo::Bar::Baz::E], resource: Photo, context: {} };
action a3 appliesTo { principal: [Photo], resource: [Document], };
action "a b", "a / b" in "a1" appliesTo {
  principal: [Photo],
  resource: [Document],
  context: { oldObject?: String }
};

types.cedarschema


// Type      := Path | SetType | RecType
// SetType   := 'Set' '<' Type '>'
// PRIMTYPE  := 'Long' | 'String' | 'Bool'
// RESERVED  := 'Bool' | 'Boolean' | 'Entity' | 'Extension' | 'Long' | 'Record' | 'Set' | 'String'
// see https://docs.cedarpolicy.com/schema/human-readable-schema-grammar.html
namespace NS1 {
  type Primitives = {
    b: Bool,
    l: Long,
    s: String,
  };
  type Sets = {
    bs: Set<Bool>,
    ls: Set<Long>,
    nested: Set<Set<String>>,
  };
  // the extension types
  type Functions = {
    da: datetime,
    de: decimal,
    du: duration,
    ip: ipaddr,
  };
  // a common type reference is not a built-in type
  type Both = {
    p: Primitives,
    f: Functions,
  };
  type Alias = Long;
  type Listy = Set<Primitives>;

  entity E {
    p: Primitives,
    q: NS1::Primitives,
  };
}
